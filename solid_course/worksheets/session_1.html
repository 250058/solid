class CafeOrder:
    """한입카페의 주문 한 건."""

    def __init__(self, customer: str) -> None:
        self.customer = customer
        self.items: list[tuple[str, int, int]] = []  # (메뉴 이름, 단가, 수량)

    def add_item(self, name: str, price: int, qty: int = 1) -> None:
        """메뉴를 주문에 담는다."""
        self.items.append((name, price, qty))

    def total(self) -> int:
        """주문 합계(원)를 계산한다."""
        return sum(price * qty for _, price, qty in self.items)

    def receipt_text(self) -> str:
        """영수증에 찍을 글자를 만든다."""
        lines = ["===== 한입카페 =====", f"손님: {self.customer}"]
        for name, price, qty in self.items:
            lines.append(f"  {name} x{qty}  {price * qty:,}원")
        lines.append(f"  합계: {self.total():,}원")
        lines.append("====================")
        return "\n".join(lines)

    def print_receipt(self) -> None:
        """영수증을 화면(프린터)으로 내보낸다."""
        print(self.receipt_text())

    def to_json(self) -> str:
        """주문을 저장용 글자(JSON)로 바꾼다."""
        data = {"customer": self.customer, "items": self.items, "total": self.total()}
        return json.dumps(data, ensure_ascii=False)

    def save(self, folder: Path) -> Path:
        """주문을 파일로 저장하고, 저장한 파일 경로를 돌려준다."""
        path = folder / f"order_{self.customer}.json"
        path.write_text(self.to_json(), encoding="utf-8")
        return path

    @classmethod
    def load(cls, path: Path) -> "CafeOrder":
        """저장된 파일을 읽어 주문을 다시 만든다."""
        data = json.loads(path.read_text(encoding="utf-8"))
        order = cls(data["customer"])
        for name, price, qty in data["items"]:
            order.add_item(name, price, qty)
        return order

📦 개념: 책임 = 변경 이유

  • 정의: 설계에서 '책임'은 하는 일의 개수가 아니라, "이 코드를 바꾸라고 요구할 수 있는 이유(그 요구를 하는 사람·부서)"이다.
  • 예: 가격 규칙(사장님), 영수증 모양(디자이너), 저장 형식(개발자) → 변경 이유 3개 = 책임 3개. 메서드가 여러 개여도 변경 이유가 하나면 책임은 하나다.

📦 개념: SRP 단일 책임 원칙 (Single Responsibility Principle)

  • 정의: 하나의 클래스(모듈)는 변경 이유가 하나뿐이어야 한다.
  • 효과: 한 요구가 한 곳만 바꾸게 되어, 다른 사람의 기능을 실수로 망가뜨릴 위험과 함께 다시 테스트할 범위가 줄어든다.

📦 개념: OCP 개방-폐쇄 원칙 (Open-Closed Principle)

  • 정의: 소프트웨어 요소는 확장에는 열려 있고, 수정에는 닫혀 있어야 한다. 즉 새 요구를 기존 코드 수정이 아니라 새 코드 추가로 받아들일 수 있어야 한다.
  • 범위: 버그 수정이나 새 객체를 만들어 넘기는 조립 코드까지 금지하는 것은 아니다. 반복해서 바뀌는 부분에 대해 기존 코드를 닫아 두는 것이다.

학교 앞 문구점의 택배 접수 프로그램입니다. 문구점 주인은 요금표를 자주 바꾸고 내년에 새 택배사도 늘릴 예정입니다. 송장 모양은 택배사 직원이, 접수 기록 저장 방식은 학교 정보 선생님이 관리합니다.

python3 s1_transfer_parcel.py를 실행해 결과를 본 뒤, 코드는 고치지 말고 읽습니다. 왼쪽 숫자는 파일의 줄 번호, # [A]~# [F]는 위치 표시입니다.

25  class ParcelService:  # [A]
26      """택배 접수 한 건을 처리한다."""
27  
28      def __init__(self, sender: str, receiver: str, carrier: str, weight_kg: float) -> None:
29          self.sender = sender
30          self.receiver = receiver
31          self.carrier = carrier
32          self.weight_kg = weight_kg
33  
34      def fee(self) -> int:  # [B]
35          """택배 요금(원)."""
36          if self.carrier == "fast":
37              return 4000 + int(self.weight_kg * 500)
38          elif self.carrier == "cheap":
39              return 2500 + int(self.weight_kg * 300)
40          elif self.carrier == "island":  # [C]
41              return 6000 + int(self.weight_kg * 800)
42          else:
43              raise ValueError(f"모르는 택배사: {self.carrier}")
44  
45      def label_text(self) -> str:  # [D]
46          """송장에 찍을 글자."""
47          if self.carrier == "fast":  # [E]
48              name = "빠른택배"
49          elif self.carrier == "cheap":
50              name = "알뜰택배"
51          else:
52              name = "섬택배"
53          return (
54              f"[{name} 송장]\n"
55              f"  보내는 사람: {self.sender}\n"
56              f"  받는 사람: {self.receiver}\n"
57              f"  무게: {self.weight_kg}kg  요금: {self.fee():,}원"
58          )
59  
60      def save_log(self, folder: Path) -> Path:  # [F]
61          """접수 기록을 파일 끝에 한 줄 덧붙인다."""
62          path = folder / "parcel_log.txt"  # Path의 / 는 경로를 이어 붙인다
63          with path.open("a", encoding="utf-8") as f:
64              f.write(f"{self.sender},{self.receiver},{self.carrier},{self.weight_kg},{self.fee()}\n")
65          return path
